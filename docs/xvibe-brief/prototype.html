const DEFAULT_BASE_URL = "https://pluggie.app/api/v1";

export function createClient(opts: ClientOptions) {
  const base = opts.baseUrl ?? DEFAULT_BASE_URL;

  async function request<T>(m: string, path: string, body?: unknown) {
    const res = await fetch(base + path, {
      method: m,
      headers: { authorization: "Bearer " + opts.token },
      body: body && JSON.stringify(body),
    });
    if (!res.ok) throw new AgentXError(res.status);
    return res.json() as Promise<T>;
  }

  return {
    leads: {
      list:   () => request<Lead[]>("GET",  "/leads"),
      create: (d: LeadInput) => request("POST", "/leads", d),
    },
    /* verifyConnection(), changes feed, uploads… */
  };
}
