// No tokens in this bundle: /api/v1 is credentialed upstream.

const api = {
  appointments: () =>
    fetch("/api/v1/appointments?sort=starts_at:asc")
      .then(r => r.json()),

  checkIn: (id) =>
    fetch(`/api/v1/appointments/${id}`, {
      method: "PATCH",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ status: "checked_in" }),
    }),
};
